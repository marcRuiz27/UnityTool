SerializedProperty representación interna que Unity hace de un campo de un objeto.
SerializedObject representación interna que Unity hace de un objeto completo

FLujo Correcto:
serializedObject.Update();          // leer estado actual
EditorGUILayout.PropertyField(...); // dibujar / modificar
serializedObject.ApplyModifiedProperties(); // guardar cambios
[CustomEditor(typeof(MiComponente))]
public class MiComponenteEditor : Editor
{
    SerializedProperty vida;

    void OnEnable()
    {
        vida = serializedObject.FindProperty("vida");
    }

    public override void OnInspectorGUI()
    {
        serializedObject.Update();

        EditorGUILayout.PropertyField(vida);

        if (GUILayout.Button("Poner vida a 0"))
        {
            vida.intValue = 0;
        }

        serializedObject.ApplyModifiedProperties();
    }
}
